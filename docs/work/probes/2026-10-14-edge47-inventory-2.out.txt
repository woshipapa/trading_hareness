== nginx feishu conf
server {
    listen 0.0.0.0:18301;
    server_name _;
    server_tokens off;

    # The listener may bind every interface so it survives a Tailscale address
    # change, but only authenticated tailnet source addresses may use it.
    allow 100.64.0.0/10;
    deny all;

    # The adapter owns both independently built SPAs. It selects the Feishu
    # dashboard for /monitor, /dashboard, /workbench and /relay, and the quant console for
    # /, /research and /personal. Keep one same-origin proxy so both apps use
    # the same API, SSE and authentication boundary.
    gzip on;
    gzip_vary on;
    gzip_comp_level 6;
    gzip_min_length 1024;
    gzip_types text/css text/javascript application/javascript application/json image/svg+xml;

    location / {
        proxy_pass http://127.0.0.1:18300;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Proto http;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_buffering off;
        proxy_read_timeout 3600s;
    }
}
